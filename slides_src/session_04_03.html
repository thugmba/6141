# 1. Input: 5 weeks of tested prices for the hoodie
import numpy as np
import pandas as pd
test = pd.read_csv("data/session_04_03_price_test.csv")

# 2. Process: a straight line through log(units) against log(price)
slope, intercept = np.polyfit(np.log(test["Test_Price_TWD"]),
                              np.log(test["Avg_Daily_Units"]), 1)
elasticity = round(slope, 2)

# 3. Output: the elasticity and what it means
print("Price elasticity of demand:", elasticity)
print(f"A 1% price rise loses about {abs(elasticity)}% of units sold.")
HOW IT WORKS
  • np.log() turns % changes into differences
  • np.polyfit(x, y, 1): the best straight line; its slope is the elasticity
OUTPUT
Price elasticity of demand: -1.55 A 1% price rise loses about 1.55% of units sold.

Step 2: Simulate Profit Before Changing a Price
 YourName_S43_step_2.pyInput → Process → Output
# 1. Input: the store's products and the elasticity from Step 1
import pandas as pd
df = pd.read_csv("data/session_04_03_dataset.csv", index_col="Product_ID")
elasticity = -1.55

# 2. Process: one function, used for every price change
def daily_profit(price_change):
    price = df["Price_TWD"] * (1 + price_change)
    units = df["Daily_Units"] * (1 + elasticity * price_change)
    return round(((price - df["Unit_Cost_TWD"]) * units).sum())

curve = {}
for change in [-0.10, -0.05, 0.0, 0.05, 0.10]:
    curve[f"{change * 100:+.0f}%"] = daily_profit(change)

# 3. Output: the whole store's daily profit at each price change
for label, profit in curve.items():
    print(f"{label}: NT${profit:,}")
HOW IT WORKS
  • daily_profit(change): new prices, new sales, total profit
  • The hoodie's elasticity is used for every product
OUTPUT
-10%: NT$17,829 -5%: NT$18,348 +0%: NT$18,620 +5%: NT$18,646 +10%: NT$18,424

Quick Check: max(curve, key=curve.get) gives +5%

# 1. Input: the store's products; the semester ends in 14 days
import pandas as pd
df = pd.read_csv("data/session_04_03_dataset.csv", index_col="Product_ID")
days_left = 14

# 2. Process: how many days the stock lasts at today's sales
df["Days_Of_Stock"] = (df["Stock_Units"] / df["Daily_Units"]).round(1)
too_slow = df[df["Days_Of_Stock"] > days_left]

# 3. Output: the products that will still have stock when the semester ends
columns = ["Product_Name", "Stock_Units", "Daily_Units", "Days_Of_Stock"]
print(too_slow[columns])
HOW IT WORKS
  • Days of stock = stock / daily sales
  • Keep the products with more than 14 days of stock
